package vm

import (
	"beedance/code"
	"beedance/compiler"
	"beedance/object"
	"fmt"
	"math"
	"unsafe"
)

// This file presents a compromise VM, named VM0. It is designed to execute
// the beedance bytecode like vm.go, but it adopts principles from vm1.go
// to be more efficient on microcontrollers when compiled with TinyGo.
//
// The key compromise is the use of a tagged value representation on the stack.
// Instead of a stack of `object.Object` interfaces, which causes significant
// garbage collector pressure and overhead from boxing, VM0 uses a stack of
// `uint64`. This allows scalar values like integers and booleans to be
// stored directly on the stack without heap allocation.
//
// Complex objects like strings, arrays, and closures are still heap-allocated,
// but the stack holds a tagged pointer to them, avoiding the interface overhead.

// --- Tagging Scheme ---
// A `Value` is a uint64. The top 16 bits are used for tagging.
// [ 16-bit tag | 48-bit payload ]

/*
The original implementation stored pointers as uintptrs in the payload of a
uint64, which is not safe for garbage collection. The GC does not scan uint64
values for pointers, so heap-allocated objects could be prematurely collected,
leading to dangling pointers and memory corruption. This is a misuse of
unsafe.Pointer.

The corrected implementation changes `Value` to a struct. This struct holds
scalars directly but uses a proper `unsafe.Pointer` for heap objects. This
ensures the Go runtime's garbage collector can trace these pointers,
preventing the underlying objects from being collected while they are still
referenced by the VM's stack.
*/
type Value struct {
	tag     uint64
	payload uint64         // For scalars (int, bool)
	ptr     unsafe.Pointer // For heap objects (string, array, etc.)
}

const (
	tagShift = 48
	// Mask to extract the payload (the lower 48 bits)
	payloadMask uint64 = 0x0000FFFFFFFFFFFF
)

// Value Tags
const (
	tagError uint64 = iota // Should not appear on stack
	tagNull
	tagBool
	tagInteger

	// Pointer types
	tagString
	tagArray
	tagHash
	tagClosure
	tagBuiltin
	tagCompiledFunc
)

// --- Value Encoding ---

func encodeInt(i int64) Value {
	// We cast the int64 to uint64 to store its bits in the payload.
	return Value{tag: tagInteger, payload: uint64(i)}
}

func encodeBool(b bool) Value {
	var payload uint64 = 0
	if b {
		payload = 1
	}
	return Value{tag: tagBool, payload: payload}
}

var encodedNull = Value{tag: tagNull}

func encodeObject(obj object.Object) Value {
	if obj == nil {
		return encodedNull
	}

	var tag uint64
	var ptr unsafe.Pointer
	switch obj.Type() {
	case object.NULL_OBJ:
		return encodedNull
	case object.BOOLEAN_OBJ:
		return encodeBool(obj.(*object.Boolean).Value)
	case object.SINT_OBJ, object.INT_OBJ, object.DINT_OBJ, object.LINT_OBJ:
		return encodeInt(obj.(*object.LInt).Value)
	case object.STRING_OBJ:
		tag = tagString
	case object.ARRAY_OBJ:
		tag = tagArray
	case object.HASH_OBJ:
		tag = tagHash
	case object.CLOSURE_OBJ:
		tag = tagClosure
	case object.BUILTIN_OBJ:
		tag = tagBuiltin
	case object.COMPILED_FUNCTION_OBJ:
		tag = tagCompiledFunc
	case object.ERROR_OBJ:
		// Errors shouldn't really be encoded, but we handle it.
		tag = tagError
	default:
		// This indicates an unhandled object type.
		return encodedNull
	}

	// For heap-allocated objects, we extract the data pointer from the interface
	// and store it in the `ptr` field, which the GC can trace.
	ptr = (*[2]unsafe.Pointer)(unsafe.Pointer(&obj))[1]
	return Value{tag: tag, ptr: ptr}
}

// --- Value Decoding ---

func (v Value) getTag() uint64 {
	return v.tag
}

func (v Value) asInt() int64 {
	// With the struct-based Value, the full int64 is stored in the payload.
	return int64(v.payload)
}

func (v Value) asBool() bool {
	return v.payload != 0
}

func (v Value) asObject() object.Object {
	tag := v.getTag()
	ptr := v.ptr

	switch tag {
	case tagNull:
		return Null
	case tagBool:
		return nativeBoolToBooleanObject(v.asBool())
	case tagInteger:
		return &object.LInt{Value: v.asInt()}
	case tagString:
		return (*object.String)(ptr)
	case tagArray:
		return (*object.Array)(ptr)
	case tagHash:
		return (*object.Hash)(ptr)
	case tagClosure:
		return (*object.Closure)(ptr)
	case tagBuiltin:
		return (*object.Builtin)(ptr)
	case tagCompiledFunc:
		return (*object.CompiledFunction)(ptr)
	case tagError:
		return (*object.Error)(ptr)
	default:
		return Null
	}
}

// VM0 is a compromise VM using tagged values on the stack.
type VM0 struct {
	constants []object.Object

	stack []Value
	sp    int // Always points to the next value. Top of stack is stack[sp-1]

	globals []object.Object

	frames      []*Frame
	framesIndex int
}

func NewVM0(bytecode *compiler.Bytecode) *VM0 {
	mainFn := &object.CompiledFunction{Instructions: bytecode.Instructions}
	mainClosure := &object.Closure{Fn: mainFn}
	mainFrame := NewFrame(mainClosure, 0)

	frames := make([]*Frame, MaxFrames)
	frames[0] = mainFrame

	return &VM0{
		constants: bytecode.Constants,

		stack: make([]Value, StackSize),
		sp:    0,

		globals: make([]object.Object, GlobalsSize),

		frames:      frames,
		framesIndex: 1,
	}
}

func (vm *VM0) LastPoppedStackElem() object.Object {
	return vm.stack[vm.sp].asObject()
}

func (vm *VM0) push(v Value) error {
	if vm.sp >= StackSize {
		return fmt.Errorf("stack overflow")
	}
	vm.stack[vm.sp] = v
	vm.sp++
	return nil
}

func (vm *VM0) pop() Value {
	v := vm.stack[vm.sp-1]
	vm.sp--
	return v
}

func (vm *VM0) currentFrame() *Frame {
	return vm.frames[vm.framesIndex-1]
}

func (vm *VM0) pushFrame(f *Frame) {
	vm.frames[vm.framesIndex] = f
	vm.framesIndex++
}

func (vm *VM0) popFrame() *Frame {
	vm.framesIndex--
	return vm.frames[vm.framesIndex]
}

func (vm *VM0) Run() error {
	var ip int
	var ins code.Instructions
	var err error

	for err == nil && vm.currentFrame().ip < len(vm.currentFrame().Instructions())-1 {
		vm.currentFrame().ip++

		ip = vm.currentFrame().ip
		ins = vm.currentFrame().Instructions()
		op := code.Opcode(ins[ip])

		switch op {
		case code.OpConstant:
			constIndex := code.ReadUint16(ins[ip+1:])
			vm.currentFrame().ip += 2
			// Constants are object.Object, we need to encode them to a Value
			err = vm.push(encodeObject(vm.constants[constIndex]))

		case code.OpPop:
			vm.pop()

		case code.OpAdd, code.OpSub, code.OpMul, code.OpDiv, code.OpMod, code.OpExponent,
			code.OpAnd, code.OpOr, code.OpXor, code.OpNand, code.OpNor:
			err = vm.executeBinaryOperation(op)

		case code.OpTrue:
			err = vm.push(encodeBool(true))

		case code.OpFalse:
			err = vm.push(encodeBool(false))

		case code.OpEqual, code.OpNotEqual, code.OpGreaterThan, code.OpLessThan,
			code.OpGreaterThanOrEqual, code.OpLessThanOrEqual:
			err = vm.executeComparison(op)

		case code.OpBang:
			err = vm.executeBangOperator()

		case code.OpMinus:
			err = vm.executeMinusOperator()

		case code.OpJump:
			pos := int(code.ReadUint16(ins[ip+1:]))
			vm.currentFrame().ip = pos - 1

		case code.OpJumpNotTruthy:
			pos := int(code.ReadUint16(ins[ip+1:]))
			vm.currentFrame().ip += 2
			condition := vm.pop()
			tag := condition.getTag()
			if tag == tagNull || (tag == tagBool && !condition.asBool()) {
				vm.currentFrame().ip = pos - 1
			}

		case code.OpNull:
			err = vm.push(encodedNull)

		case code.OpSetGlobal:
			globalIndex := code.ReadUint16(ins[ip+1:])
			vm.currentFrame().ip += 2
			vm.globals[globalIndex] = vm.pop().asObject()

		case code.OpGetGlobal:
			globalIndex := code.ReadUint16(ins[ip+1:])
			vm.currentFrame().ip += 2
			err = vm.push(encodeObject(vm.globals[globalIndex]))

		case code.OpSetLocal:
			localIndex := code.ReadUint8(ins[ip+1:])
			vm.currentFrame().ip += 1
			frame := vm.currentFrame()
			vm.stack[frame.basePointer+int(localIndex)] = vm.pop()

		case code.OpGetLocal:
			localIndex := code.ReadUint8(ins[ip+1:])
			vm.currentFrame().ip += 1
			frame := vm.currentFrame()
			err = vm.push(vm.stack[frame.basePointer+int(localIndex)])

		default:
			// Opcodes for Array, Hash, Call, ReturnValue, Closure, etc. are
			// not implemented in this example but would follow a similar pattern:
			// pop values, check tags, decode to objects if necessary, perform
			// the operation, encode the result, and push it back.
			return fmt.Errorf("opcode %d (%d) not implemented in VM0", op, op)
		}
	}

	return err
}

func (vm *VM0) executeBinaryBooleanOperation(
	op code.Opcode,
	left, right Value,
) error {
	leftValue := left.asBool()
	rightValue := right.asBool()

	var result bool
	switch op {
	case code.OpAnd:
		result = leftValue && rightValue
	case code.OpOr:
		result = leftValue || rightValue
	case code.OpXor:
		result = leftValue != rightValue
	case code.OpNand:
		result = !(leftValue && rightValue)
	case code.OpNor:
		result = !(leftValue || rightValue)
	default:
		return fmt.Errorf("unknown boolean operator: %d", op)
	}
	return vm.push(encodeBool(result))
}

func (vm *VM0) executeBinaryOperation(op code.Opcode) error {
	rightV := vm.pop()
	leftV := vm.pop()

	// Fast path for booleans
	if leftV.getTag() == tagBool && rightV.getTag() == tagBool {
		return vm.executeBinaryBooleanOperation(op, leftV, rightV)
	}

	// Fast path for integers
	if leftV.getTag() == tagInteger && rightV.getTag() == tagInteger {
		left := leftV.asInt()
		right := rightV.asInt()
		var result int64
		switch op {
		case code.OpAdd:
			result = left + right
		case code.OpSub:
			result = left - right
		case code.OpMul:
			result = left * right
		case code.OpDiv:
			if right == 0 {
				return fmt.Errorf("division by zero")
			}
			result = left / right
		case code.OpMod:
			if right == 0 {
				return fmt.Errorf("division by zero")
			}
			result = left % right
		case code.OpExponent:
			result = int64(math.Pow(float64(left), float64(right)))
		case code.OpAnd:
			result = left & right
		case code.OpOr:
			result = left | right
		case code.OpXor:
			result = left ^ right
		case code.OpNand:
			result = ^(left & right)
		case code.OpNor:
			result = ^(left | right)
		default:
			return fmt.Errorf("unknown integer operator: %d", op)
		}
		return vm.push(encodeInt(result))
	}

	// Slow path: decode to objects and use existing logic
	leftObj := leftV.asObject()
	rightObj := rightV.asObject()

	if leftObj.Type() == object.STRING_OBJ && rightObj.Type() == object.STRING_OBJ {
		if op != code.OpAdd {
			return fmt.Errorf("unknown string operator: %d", op)
		}
		left := leftObj.(*object.String).Value
		right := rightObj.(*object.String).Value
		result := &object.String{Value: left + right}
		return vm.push(encodeObject(result))
	}

	return fmt.Errorf("unsupported types for binary operation: %s %d %s",
		leftObj.Type(), op, rightObj.Type())
}

func (vm *VM0) executeComparison(op code.Opcode) error {
	rightV := vm.pop()
	leftV := vm.pop()

	// Fast path for integers
	if leftV.getTag() == tagInteger && rightV.getTag() == tagInteger {
		left := leftV.asInt()
		right := rightV.asInt()
		switch op {
		case code.OpEqual:
			return vm.push(encodeBool(left == right))
		case code.OpNotEqual:
			return vm.push(encodeBool(left != right))
		case code.OpGreaterThan:
			return vm.push(encodeBool(left > right))
		case code.OpLessThan:
			return vm.push(encodeBool(left < right))
		case code.OpGreaterThanOrEqual:
			return vm.push(encodeBool(left >= right))
		case code.OpLessThanOrEqual:
			return vm.push(encodeBool(left <= right))
		default:
			return fmt.Errorf("unknown integer comparison: %d", op)
		}
	}

	// Fast path for booleans
	if leftV.getTag() == tagBool && rightV.getTag() == tagBool {
		left := leftV.asBool()
		right := rightV.asBool()
		switch op {
		case code.OpEqual:
			return vm.push(encodeBool(left == right))
		case code.OpNotEqual:
			return vm.push(encodeBool(left != right))
		default:
			return fmt.Errorf("unsupported comparison for booleans: %d", op)
		}
	}

	// Slow path for object comparison
	leftObj := leftV.asObject()
	rightObj := rightV.asObject()
	switch op {
	case code.OpEqual:
		return vm.push(encodeBool(leftObj == rightObj))
	case code.OpNotEqual:
		return vm.push(encodeBool(leftObj != rightObj))
	default:
		return fmt.Errorf("unsupported comparison: %s %d %s", leftObj.Type(), op, rightObj.Type())
	}
}

func (vm *VM0) executeBangOperator() error {
	operand := vm.pop()
	tag := operand.getTag()
	if tag == tagNull || (tag == tagBool && !operand.asBool()) {
		return vm.push(encodeBool(true))
	}
	return vm.push(encodeBool(false))
}

func (vm *VM0) executeMinusOperator() error {
	operand := vm.pop()
	if operand.getTag() != tagInteger {
		return fmt.Errorf("unsupported type for negation: not an integer")
	}
	value := operand.asInt()
	return vm.push(encodeInt(-value))
}
